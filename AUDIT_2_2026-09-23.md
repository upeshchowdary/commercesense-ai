### A. Grounding: require the snippet to actually appear in findings (`agent/insight_agent.py`)

```python
def _norm(s: str) -> str:
    return " ".join(s.split()).lower()


def _apply_source_match_guardrail(
    insights: list[Insight], bundle: ResearchBundle
) -> list[Insight]:
    """Rule 4: a cited snippet must literally appear in one of Agent 5's
    findings. Anything else is stripped to null and logged."""
    corpus = [(_norm(f.snippet), f.source_url) for f in bundle.findings]
    for insight in insights:
        if insight.confidence == "none" or not insight.source_snippet:
            continue
        needle = _norm(insight.source_snippet)
        match = next((url for text, url in corpus if needle and needle in text), None)
        if match is None:
            log_decision(
                agent="insight_agent",
                product_name=bundle.product_name,
                action="grounding_strip",
                detail={"category": insight.category, "snippet": insight.source_snippet[:200]},
            )
            insight.summary = None
            insight.confidence = "none"
            insight.source_snippet = None
            insight.source_url = None
        else:
            insight.source_url = match  # trust the finding's URL, not the model's
    return insights
```

In `run_insight_agent`, call it right after `_apply_grounding_guardrail(insights)`:

```python
    insights = _apply_grounding_guardrail(insights)
    insights = _apply_source_match_guardrail(insights, bundle)
```
